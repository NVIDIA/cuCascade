cmake_minimum_required(VERSION 4.0)
project(pinned_pr_harness LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
find_package(CUDAToolkit REQUIRED)
find_package(rmm REQUIRED CONFIG)
find_package(cudf REQUIRED CONFIG)
find_package(Threads REQUIRED)
add_library(allocator_lib SHARED IMPORTED)
set_target_properties(allocator_lib PROPERTIES IMPORTED_LOCATION "${ALLOC_BINARY}/libcucascade.so" INTERFACE_INCLUDE_DIRECTORIES "${ALLOC_SOURCE}/include")
add_executable(pinned_bench pinned_bench.cpp)
target_compile_options(pinned_bench PRIVATE -O3 -Wall -Wextra)
target_link_libraries(pinned_bench PRIVATE allocator_lib "${ALLOC_BINARY}/libcucascade_topology_discovery.so" cudf::cudf rmm::rmm CUDA::cudart Threads::Threads)
if(CANDIDATE)
 target_compile_definitions(pinned_bench PRIVATE PR_CANDIDATE=1)
endif()
